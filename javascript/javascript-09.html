// Program 9: Palindrome Number let num = 121; let original = num; let reverse = 0; while (num > 0) { let digit = num % 10; reverse = reverse * 10 + digit; num = Math.floor(num / 10); } if (original === reverse) { console.log(original + " is a Palindrome"); } else { console.log(original + " is not a Palindrome"); }
